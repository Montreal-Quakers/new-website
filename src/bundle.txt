<script>
  async function initPagefind() {
    const container = document.getElementById('pagefind-search-ui');
    if (!container) return;

    // 1. Wipe container content completely
    container.innerHTML = '';

    // 2. Nuke global Pagefind cache so it doesn't hold stale language shards in SPA memory
    if ('pagefind' in window) {
      delete (window as any).pagefind;
    }
    if ('PagefindUI' in window) {
      delete (window as any).PagefindUI;
    }

    const currentLang = document.documentElement.lang || container.dataset.lang || 'en';
    const urlParams = new URLSearchParams(window.location.search);
    const query = urlParams.get('q');

    const config = {
      element: "#pagefind-search-ui",
      bundlePath: "/pagefind/",
      autofocus: true,
      focusOnSlash: true,
      showImages: false,
      showFilters: true,
      showEmptyFilters: true,
      translations: { 
        placeholder: currentLang === 'fr' ? "Appuyez sur / pour chercher" : "Press / to search" 
      }
    };

    // 3. Dynamically fetch fresh script to bypass module/instance caching across view transitions
    const scriptTag = document.createElement('script');
    scriptTag.src = '/pagefind/pagefind-ui.js';
    scriptTag.async = true;
    
    scriptTag.onload = () => {
      if (typeof (window as any).PagefindUI !== 'undefined') {
        const ui = new (window as any).PagefindUI(config);
        
        if (query) {
          setTimeout(() => ui.triggerSearch(query), 150);
        }

        const searchInput = container.querySelector<HTMLInputElement>('.pagefind-ui__search-input');
        if (searchInput) {
          searchInput.addEventListener('input', (e) => {
            const target = e.target as HTMLInputElement;
            const val = target.value.trim();
            const url = new URL(window.location.href);

            if (val) {
              url.searchParams.set('q', val);
            } else {
              url.searchParams.delete('q');
            }

            window.history.replaceState({}, '', url.toString());
          });
        }
      }
    };

    document.head.appendChild(scriptTag);

    if (!document.querySelector('link[href="/pagefind/pagefind-ui.css"]')) {
      const link = document.createElement('link');
      link.rel = 'stylesheet';
      link.href = '/pagefind/pagefind-ui.css';
      document.head.appendChild(link);
    }
  }

  // Bind to Astro's view transition page load event
  document.addEventListener('astro:page-load', initPagefind);
</script>

// src/pages/[...slug].astro
import { getCollection, render } from 'astro:content';
import Layout from '../layouts/Layout.astro';
import { Icon } from 'astro-icon/components';
import { Image, Picture } from 'astro:assets';
import ContactForm from '../components/ContactForm.astro';
import Calendar from '../components/calendar.astro';
import Calendar2 from '../components/calendar2.astro';
import NextSteps from '../components/NextSteps.astro';
import AutoLink from '../components/AutoLink.astro';

export async function getStaticPaths() {
  const allPages = await getCollection('pages');
  
  // Filter out intro fragments so Astro doesn't build standalone pages for them
  const standalonePages = allPages.filter(entry => entry.data?.translationID !== 'news');

  return standalonePages.map(entry => {
    // 1. Force the ID to be a string
    // 2. Remove the file extension entirely
    // 3. Remove 'index' if it's at the end
    const cleanSlug = entry.id
      .replace(/\.(mdx|md|html)$/, '')
      .replace(/\/index$/, '');

    // If the file was just 'index.mdx', cleanSlug is now empty. 
    // Astro needs 'undefined' for the root index in [...slug]
    const finalSlug = cleanSlug === 'index' || cleanSlug === '' ? undefined : cleanSlug;

    return {
      params: { slug: finalSlug }, 
      props: { entry },
    }
  });
}

const { entry } = Astro.props;
const { Content } = await render(entry);

// Adding components here ensures they work inside your MDX files
const components = { 
  a: AutoLink,
  Icon, 
  Image, 
  Picture,
  ContactForm, 
  NextSteps, 
  Calendar,
  Calendar2 
};
